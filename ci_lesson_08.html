บทนี้เป็นบทสุดท้ายของเนื้อหา และเป็นฐานของงานกลุ่มครั้งที่ 3 ซึ่งมีน้ำหนักร้อยละ 25 ของคะแนนรายวิชา รูบริกที่ 3 ให้น้ำหนักการประเมินโอกาสทางนวัตกรรมร้อยละ 20 และกำหนดว่าระดับสูงสุดต้องใช้กรอบการประเมินที่เรียน เช่น ระดับความพร้อมของเทคโนโลยี ขนาดตลาด และคู่แข่ง พร้อมข้อมูลที่มีที่มา เนื้อหาในหัวข้อ 8.2 ถึง 8.4 จึงเป็นสิ่งที่กลุ่มต้องใช้โดยตรง

8.1 โอกาสทางนวัตกรรมที่เกิดจากเทคโนโลยี

8.1.1 เทคโนโลยีผลักกับตลาดดึง

นวัตกรรมเริ่มต้นได้จาก 2 ทาง ทางแรกคือ เทคโนโลยีผลัก (Technology push) หมายถึงนวัตกรรมที่เริ่มจากความรู้หรือเทคโนโลยีใหม่ แล้วจึงหาว่าใช้แก้ปัญหาอะไรได้ ทางที่สองคือ ตลาดดึง (Market pull หรือ Demand pull) หมายถึงนวัตกรรมที่เริ่มจากความต้องการหรือปัญหาของลูกค้า แล้วจึงหาเทคโนโลยีมาตอบ กระบวนการคิดเชิงออกแบบในบทที่ 3 เป็นแนวทางแบบตลาดดึง เพราะเริ่มจากปัญหาของผู้ใช้ ส่วนผลงานวิจัยของมหาวิทยาลัยในบทที่ 6 มักเป็นแบบเทคโนโลยีผลัก ตารางที่ 8.1 เปรียบเทียบทั้งสองทาง

ตารางที่ 8.1 เทคโนโลยีผลักกับตลาดดึง

ประเด็นเทคโนโลยีผลักตลาดดึง
จุดเริ่มต้นความรู้หรือเทคโนโลยีใหม่ปัญหาหรือความต้องการของลูกค้า
คำถามแรกเทคโนโลยีนี้ใช้ทำอะไรได้บ้างลูกค้ามีปัญหาอะไรที่ยังไม่มีใครแก้ได้ดี
จุดแข็งสร้างสิ่งที่ตลาดยังนึกไม่ถึง และเลียนแบบได้ยากตรงกับความต้องการจริง ขายได้ง่ายกว่า
ความเสี่ยงหลักไม่มีลูกค้าที่ต้องการจริงได้นวัตกรรมที่เลียนแบบได้ง่าย
ตัวอย่างนักวิจัยพัฒนาวัสดุเคลือบผิวใหม่ แล้วหาว่าใช้กับสินค้าใดร้านค้าพบว่าลูกค้ารอคิวนาน จึงหาระบบจองคิว

Di Stefano และคณะ (2012) ทบทวนงานวิจัยด้านนวัตกรรมที่ศึกษาสองแนวคิดนี้ และสรุปว่าแนวคิดทั้งสองไม่ได้ขัดกัน นวัตกรรมเกิดจากการเชื่อมความสามารถทางเทคโนโลยีเข้ากับความต้องการของตลาด ข้อสรุปเชิงปฏิบัติคือ กลุ่มที่เริ่มจากปัญหาของผู้ใช้ควรถามต่อว่ามีเทคโนโลยีใดที่ทำให้แก้ปัญหาได้ดีกว่าหรือถูกกว่าเดิม ส่วนกลุ่มที่เริ่มจากเทคโนโลยีต้องหาหลักฐานว่ามีผู้ใช้ที่มีปัญหาจริง ด้วยวิธีในบทที่ 3 และ 4

8.1.2 แหล่งของโอกาสจากเทคโนโลยี

โอกาสทางนวัตกรรมจากเทคโนโลยีไม่ได้มาจากเทคโนโลยีที่ใหม่ที่สุดเสมอไป สำหรับธุรกิจขนาดเล็กและนวัตกรรมของนิสิต โอกาสมักมาจากเทคโนโลยีที่มีอยู่แล้วแต่เพิ่งถูกลงหรือใช้ง่ายขึ้น ตารางที่ 8.2 สรุปแหล่งของโอกาสที่พบบ่อย

ตารางที่ 8.2 แหล่งของโอกาสทางนวัตกรรมจากเทคโนโลยี

แหล่งของโอกาสความหมายตัวอย่าง (กรณีสมมติเพื่อการสอน)
เทคโนโลยีที่ราคาลดลงสิ่งที่เคยแพงเกินกว่าธุรกิจขนาดเล็กจะใช้ได้ เริ่มมีราคาที่รับได้ร้านอาหารใช้เซนเซอร์วัดอุณหภูมิตู้แช่ราคาไม่แพง แจ้งเตือนผ่านโทรศัพท์
เทคโนโลยีที่ใช้ง่ายขึ้นเครื่องมือที่เคยต้องใช้ผู้เชี่ยวชาญ คนทั่วไปใช้ได้ร้านค้าทำแบบฟอร์มสั่งสินค้าออนไลน์เองโดยไม่ต้องเขียนโปรแกรม
เทคโนโลยีจากอุตสาหกรรมอื่นนำวิธีที่ใช้ในอุตสาหกรรมหนึ่งไปใช้อีกอุตสาหกรรมนำระบบติดตามพัสดุมาใช้ติดตามผ้าที่ส่งซัก
สิทธิบัตรที่หมดอายุการประดิษฐ์ที่ใครก็ใช้ได้แล้ว (บทที่ 6)ปรับกลไกจากสิทธิบัตรเก่าให้เหมาะกับผลิตภัณฑ์ชุมชน
ผลงานวิจัยที่ยังไม่มีผู้นำไปใช้เทคโนโลยีในมหาวิทยาลัยที่พร้อมถ่ายโอน (บทที่ 6)วิสาหกิจชุมชนขออนุญาตใช้วิธียืดอายุอาหาร
แพลตฟอร์มดิจิทัลที่มีผู้ใช้จำนวนมากใช้แพลตฟอร์มที่ลูกค้าใช้อยู่แล้วเป็นช่องทางส่งรหัสผ่านแอปพลิเคชันส่งข้อความที่ผู้ใช้มีอยู่

8.1.3 คำถามสำหรับแปลงเทคโนโลยีเป็นโอกาส

เทคโนโลยีจะเป็นโอกาสทางนวัตกรรมได้ ต้องผ่านคำถามอย่างน้อย 4 ข้อ ข้อแรก เทคโนโลยีนี้ทำอะไรได้ดีกว่าวิธีเดิม เช่น เร็วกว่า ถูกกว่า หรือแม่นยำกว่า ข้อที่สอง ใครได้ประโยชน์จากความดีกว่านั้นมากที่สุด ข้อที่สาม ผู้ได้ประโยชน์ยอมจ่ายเงินหรือเปลี่ยนพฤติกรรมเพื่อสิ่งนี้หรือไม่ และข้อที่สี่ ทีมเข้าถึงเทคโนโลยีนี้ได้หรือไม่ เช่น ซื้อได้ ขออนุญาตใช้ได้ หรือพัฒนาเองได้ ถ้าตอบข้อที่สามไม่ได้ เทคโนโลยีนั้นยังเป็นเพียงความเป็นไปได้ ไม่ใช่โอกาส หัวข้อ 8.2 ถึง 8.4 เป็นเครื่องมือสำหรับตอบคำถามเหล่านี้อย่างเป็นระบบ

8.2 ระดับความพร้อมของเทคโนโลยี

8.2.1 ความหมายและที่มา

ระดับความพร้อมของเทคโนโลยี (Technology Readiness Level หรือ TRL) คือมาตรวัดความพร้อมของเทคโนโลยีหนึ่ง ๆ แบ่งเป็น 9 ระดับ ระดับ 1 ต่ำสุด และระดับ 9 สูงสุด องค์การบริหารการบินและอวกาศแห่งชาติของสหรัฐอเมริกา (NASA) อธิบายว่าโครงการเทคโนโลยีแต่ละโครงการจะถูกประเมินเทียบกับเกณฑ์ของแต่ละระดับ แล้วกำหนดระดับตามความก้าวหน้าของโครงการ (National Aeronautics and Space Administration [NASA], 2023) Mankins (2009) ทบทวนความเป็นมาของมาตรวัดนี้ ซึ่งพัฒนาขึ้นที่ NASA และต่อมามีหน่วยงานอื่นนำไปใช้ในการประเมินเทคโนโลยีอย่างแพร่หลาย

ประโยชน์ของ TRL คือทำให้คนที่มีพื้นฐานต่างกัน เช่น นักวิจัย นักลงทุน และผู้ให้ทุน ใช้ภาษาเดียวกันเมื่อพูดถึงความพร้อมของเทคโนโลยี และช่วยให้เห็นว่าเทคโนโลยียังต้องผ่านขั้นใดอีกบ้างก่อนนำไปใช้จริง

8.2.2 ระดับทั้ง 9 ระดับ

ตารางที่ 8.3 สรุปความหมายของแต่ละระดับตามคำอธิบายของ NASA (2023) ในคอลัมน์ที่สอง คอลัมน์ที่สามเป็นถ้อยคำที่ผู้เขียนปรับสำหรับนวัตกรรมทางธุรกิจเพื่อใช้ในการสอน ไม่ใช่นิยามทางการ เพราะนิยามของ NASA เขียนสำหรับเทคโนโลยีอวกาศ เช่น ระดับ 7 ถึง 9 อ้างถึงการทดสอบในสภาพแวดล้อมอวกาศและภารกิจการบิน

ตารางที่ 8.3 ระดับความพร้อมของเทคโนโลยี 9 ระดับ

ระดับความหมายตามคำอธิบายของ NASAถ้อยคำปรับสำหรับนวัตกรรมทางธุรกิจ (เพื่อการสอน)
1เริ่มการวิจัยทางวิทยาศาสตร์ และแปลผลไปสู่การวิจัยและพัฒนาในอนาคตมีความรู้หรือหลักการพื้นฐานที่อาจนำมาใช้ได้
2ศึกษาหลักการพื้นฐานแล้ว และเริ่มคิดการประยุกต์ใช้ ยังแทบไม่มีหลักฐานจากการทดลองมีแนวคิดว่าจะนำหลักการไปใช้แก้ปัญหาอะไร แต่ยังไม่ได้ลอง
3เริ่มวิจัยและออกแบบอย่างจริงจัง ศึกษาเชิงวิเคราะห์และในห้องปฏิบัติการ มักสร้างแบบจำลองพิสูจน์แนวคิดทำต้นแบบพิสูจน์แนวคิด แสดงว่าหลักการใช้ได้
4ทดสอบชิ้นส่วนหลายชิ้นร่วมกันประกอบส่วนสำคัญเข้าด้วยกันและทดสอบในสถานที่ของทีม
5ทดสอบเข้มข้นกว่าระดับ 4 ในสภาพที่ใกล้เคียงสภาพจริงมากที่สุดทดสอบกับผู้ใช้ในสถานการณ์ที่ใกล้เคียงการใช้จริง
6มีต้นแบบหรือแบบจำลองที่ทำงานได้ครบทุกหน้าที่ต้นแบบทำงานได้ครบทุกหน้าที่ และทดสอบกับผู้ใช้จริงแล้ว
7สาธิตต้นแบบที่ทำงานได้ในสภาพแวดล้อมจริงของการใช้งาน (สำหรับ NASA คืออวกาศ)ทดลองให้บริการหรือขายจริงในพื้นที่เล็ก
8ทดสอบและรับรองแล้ว พร้อมนำไปใช้ในระบบที่มีอยู่ผ่านข้อกำหนดและมาตรฐานที่เกี่ยวข้อง พร้อมผลิตหรือให้บริการในขนาดใหญ่ขึ้น
9ผ่านการใช้งานจริงในภารกิจที่สำเร็จแล้วขายหรือให้บริการจริงต่อเนื่อง และมีลูกค้าที่ทีมไม่รู้จักมาก่อน

8.2.3 การระบุระดับของนวัตกรรมของกลุ่ม

การระบุระดับ TRL ต้องมีหลักฐาน ไม่ใช่ความรู้สึกของทีม หลักการคือ นวัตกรรมอยู่ในระดับใดระดับหนึ่งก็ต่อเมื่อมีหลักฐานครบสำหรับระดับนั้น ถ้ามีหลักฐานบางส่วนของระดับที่สูงกว่า ให้ระบุระดับที่มีหลักฐานครบ แล้วบอกว่ากำลังดำเนินการในระดับถัดไป

นวัตกรรมของกลุ่มส่วนใหญ่ที่ผ่านงานกลุ่มครั้งที่ 2 มาแล้วจะอยู่ระหว่างระดับ 4 ถึง 6 ตามถ้อยคำปรับในตารางที่ 8.3 เพราะมีต้นแบบและผลการทดสอบกับผู้ใช้ แต่ยังไม่ได้ขายจริง ตารางที่ 8.4 แสดงตัวอย่างการบันทึก

ตารางที่ 8.4 ตัวอย่างการระบุระดับความพร้อมของเทคโนโลยี (บริการตู้รับผ้า กรณีสมมติเพื่อการสอน)

ส่วนของนวัตกรรมระดับที่มีหลักฐานครบหลักฐานสิ่งที่ต้องทำเพื่อขึ้นระดับถัดไป
ตู้ล็อกและกลไกเปิดปิด6ใช้ตู้ที่มีขายในตลาด ทดสอบกับผู้ใช้ 10 คน 2 สัปดาห์ทดลองให้บริการจริงที่หอพัก 1 แห่ง เก็บเงินจริง
ระบบส่งรหัสครั้งเดียว5ทีมส่งรหัสด้วยมือระหว่างทดสอบ ยังไม่เป็นระบบอัตโนมัติทำระบบส่งรหัสอัตโนมัติและทดสอบกับผู้ใช้
การประสานงานกับร้านซักอบรีด4ร้าน 1 แห่งรับผ้าจากตู้ในช่วงทดสอบทดสอบการรับส่ง 1 เดือนโดยไม่มีทีมช่วย

ตัวอย่างนี้แสดงว่านวัตกรรมหนึ่งชิ้นอาจมีหลายส่วนที่ความพร้อมไม่เท่ากัน ความพร้อมของนวัตกรรมทั้งชิ้นถูกจำกัดด้วยส่วนที่พร้อมน้อยที่สุด ในตัวอย่างคือการประสานงานกับร้าน ซึ่งไม่ใช่เทคโนโลยีด้วยซ้ำ ทีมจึงควรให้ความสำคัญกับส่วนนี้ก่อน

8.2.4 ข้อจำกัดของ TRL

TRL วัดความพร้อมของเทคโนโลยี ไม่ได้วัดความพร้อมของตลาดหรือธุรกิจ เทคโนโลยีที่อยู่ระดับ 9 อาจไม่มีลูกค้าก็ได้ ในทางกลับกัน บริการที่ใช้เทคโนโลยีธรรมดาซึ่งพร้อมใช้อยู่แล้ว อาจล้มเหลวเพราะไม่มีผู้ใช้ ด้วยเหตุนี้ รูบริกที่ 3 จึงกำหนดให้ประเมินทั้งความพร้อมของเทคโนโลยี ขนาดตลาด และคู่แข่ง ไม่ใช่ TRL อย่างเดียว นอกจากนี้ นวัตกรรมที่เป็นบริการหรือโมเดลธุรกิจ เช่น บริการรับฝากของในชุมชน อาจไม่มีเทคโนโลยีใหม่เลย กรณีนี้กลุ่มใช้ถ้อยคำปรับในตารางที่ 8.3 กับตัวบริการแทน และระบุให้ชัดว่าเป็นการปรับใช้

8.3 การประเมินโอกาสทางตลาด

การประเมินโอกาสทางตลาดตอบคำถามว่า ถ้าเทคโนโลยีพร้อม จะมีลูกค้ามากพอให้ธุรกิจอยู่ได้หรือไม่ ในรายวิชานี้แบ่งการประเมินเป็น 3 ส่วน ได้แก่ ขนาดตลาด ความพร้อมของลูกค้า และคู่แข่งกับทางเลือกเดิม

8.3.1 ขนาดตลาด

ขนาดตลาดประมาณได้ 2 วิธี วิธีแรกคือ การประมาณจากบนลงล่าง (Top-down) เริ่มจากตัวเลขของตลาดใหญ่ เช่น มูลค่าตลาดทั้งประเทศจากรายงาน แล้วคูณด้วยสัดส่วนที่เกี่ยวข้อง วิธีที่สองคือ การประมาณจากล่างขึ้นบน (Bottom-up) เริ่มจากหน่วยเล็กที่สุดที่ทีมรู้จริง เช่น จำนวนลูกค้าในพื้นที่ คูณด้วยสัดส่วนที่ซื้อ ความถี่ และราคา

สำหรับนวัตกรรมของนิสิต วิธีจากล่างขึ้นบนเหมาะกว่า เพราะใช้ข้อมูลที่ทีมเก็บได้เองจากบทที่ 3 และ 4 และทำให้เห็นว่าตัวเลขแต่ละตัวมาจากไหน ส่วนวิธีจากบนลงล่างมักให้ตัวเลขที่ใหญ่เกินจริง เพราะสมมติว่าทีมจะได้ส่วนแบ่งจากตลาดทั้งประเทศ ซึ่งทีมเข้าถึงไม่ได้ ตารางที่ 8.5 แสดงตัวอย่างการคำนวณ

ตารางที่ 8.5 ตัวอย่างการประมาณขนาดตลาดแบบจากล่างขึ้นบน (บริการตู้รับผ้า กรณีสมมติเพื่อการสอน)

ขั้นตัวเลขที่มาของตัวเลขผลลัพธ์
1. จำนวนห้องพักในพื้นที่40 หอพัก × 80 ห้องนับจากการสำรวจพื้นที่รอบมหาวิทยาลัย3,200 ห้อง
2. สัดส่วนที่มีปัญหาส่งผ้าไม่ทันร้อยละ 30ผลการสัมภาษณ์ในบทที่ 3960 คน
3. สัดส่วนที่ยอมจ่ายร้อยละ 50ผลการทดสอบในบทที่ 4480 คน
4. การใช้ต่อคนต่อเดือน4 ครั้ง × 20 บาทผลการทดสอบ และราคาในบทที่ 780 บาทต่อคนต่อเดือน
5. ตลาดที่เข้าถึงได้ต่อเดือน480 × 80คำนวณ38,400 บาทต่อเดือน
6. ส่วนที่คาดว่าจะได้ในปีแรกร้อยละ 25 ของขั้นที่ 5ทีมติดตั้งตู้ได้ 10 จาก 40 หอพัก9,600 บาทต่อเดือน

ตัวเลขในขั้นที่ 6 คือรายได้ที่ทีมคาดว่าจะได้จริงในปีแรก ซึ่งเล็กกว่าขั้นที่ 5 มาก ตัวเลขนี้ต้องนำไปเทียบกับต้นทุนในบทที่ 7 ถ้าต้องติดตั้ง 10 ตู้ ราคาตู้ละ 24,000 บาท เงินลงทุนจะเป็น 240,000 บาท ซึ่งมากกว่ารายได้ต่อเดือนหลายเท่า ทีมจึงต้องคำนวณระยะเวลาคืนทุนใหม่ และพิจารณาว่าควรเริ่มด้วยจำนวนตู้น้อยกว่านี้หรือไม่

ข้อสำคัญที่สุดของตารางนี้คือคอลัมน์ที่มาของตัวเลข รูบริกที่ 3 กำหนดว่าระดับสูงสุดต้องมีข้อมูลที่มีที่มา ตัวเลขที่ทีมคาดเดาเองต้องระบุว่าเป็นการคาดเดา และควรมีกรณีที่แย่กว่าที่คาดประกอบ เช่นเดียวกับการคำนวณในบทที่ 7

8.3.2 ความพร้อมของลูกค้า

ลูกค้าที่มีปัญหาอาจยังไม่พร้อมใช้นวัตกรรมก็ได้ ความพร้อมของลูกค้าดูได้จาก 3 ด้าน ด้านแรกคือลูกค้ารู้ตัวหรือไม่ว่ามีปัญหา ถ้าลูกค้าไม่รู้สึกว่าเป็นปัญหา ทีมต้องใช้ทรัพยากรมากในการสื่อสาร ด้านที่สองคือลูกค้าพยายามแก้ปัญหาเองอยู่แล้วหรือไม่ ถ้าพยายามอยู่ แปลว่าปัญหารุนแรงพอ ด้านที่สามคือลูกค้าต้องเปลี่ยนพฤติกรรมมากเพียงใดเพื่อใช้นวัตกรรม ยิ่งต้องเปลี่ยนมาก การยอมรับยิ่งช้า ตามคุณลักษณะของนวัตกรรมที่ Rogers (2003) อธิบายไว้และบทที่ 5 และ 7 ได้กล่าวถึงแล้ว

8.3.3 คู่แข่งและทางเลือกเดิม

นิสิตมักเขียนว่านวัตกรรมของกลุ่ม "ไม่มีคู่แข่ง" ซึ่งแทบไม่เคยเป็นจริง เพราะลูกค้าที่มีปัญหาย่อมมีวิธีจัดการกับปัญหาอยู่แล้ว แม้จะเป็นวิธีที่ไม่ดีนัก วิธีเหล่านี้เรียกว่า ทางเลือกเดิม (Existing alternatives) และเป็นคู่แข่งที่แท้จริงของนวัตกรรม ตารางที่ 8.6 แสดงการเปรียบเทียบนวัตกรรมกับทางเลือกเดิมและคู่แข่งโดยตรง

ตารางที่ 8.6 ตัวอย่างการเปรียบเทียบกับคู่แข่งและทางเลือกเดิม (บริการตู้รับผ้า กรณีสมมติเพื่อการสอน)

ทางเลือกราคาต่อครั้งความสะดวกด้านเวลาจุดอ่อนลูกค้าที่ยังเลือกทางนี้
ไปส่งที่ร้านเองในวันหยุดค่าเดินทางต้องรอวันหยุดผ้าค้างหลายวันผู้ที่มีเวลาว่างวันหยุด
ฝากเพื่อนส่งไม่มีค่าใช้จ่ายขึ้นกับเพื่อนเกรงใจ ไม่แน่นอนผู้ที่มีเพื่อนกลับเร็ว
ร้านที่มีบริการรับส่งถึงห้อง30 บาทนัดเวลาได้ต้องอยู่ห้องตอนรับส่งผู้ที่อยู่ห้องเป็นเวลา
เครื่องซักผ้าหยอดเหรียญในหอพักค่าซักต่อรอบซักได้ทุกเวลาต้องซักและตากเองผู้ที่มีเวลาและที่ตาก
บริการตู้รับผ้า20 บาทฝากรับได้ตลอด 24 ชั่วโมงต้องมีตู้ในหอพักของตนผู้ที่กลับหลังร้านปิดและไม่อยากซักเอง

คอลัมน์สุดท้ายสำคัญที่สุด เพราะบอกว่าลูกค้ากลุ่มใดจะยังไม่เปลี่ยนมาใช้นวัตกรรม แม้นวัตกรรมจะดีกว่าในบางด้าน ตารางนี้ช่วยให้ทีมกำหนดกลุ่มผู้ใช้แรกได้แคบและชัดขึ้น ตามหลักในบทที่ 5

8.4 การประเมินความเสี่ยงของนวัตกรรม

8.4.1 ความเสี่ยง 5 ด้าน

ความเสี่ยงของนวัตกรรมคือเหตุการณ์ที่อาจเกิดขึ้นและทำให้นวัตกรรมไม่สำเร็จตามแผน ในรายวิชานี้จำแนกความเสี่ยงเป็น 5 ด้าน ตามตารางที่ 8.7 การจำแนกนี้ช่วยให้ทีมไม่มองข้ามด้านที่ไม่คุ้นเคย เช่น ทีมที่ถนัดเทคโนโลยีมักมองข้ามความเสี่ยงด้านตลาด และทีมที่ถนัดการตลาดมักมองข้ามความเสี่ยงด้านเทคโนโลยี

ตารางที่ 8.7 ความเสี่ยงของนวัตกรรม 5 ด้าน

ด้านคำถามหลักตัวอย่างความเสี่ยงหลักฐานที่ลดความไม่แน่นอน
เทคโนโลยีนวัตกรรมทำงานได้ตามที่ออกแบบทุกครั้งหรือไม่ระบบส่งรหัสล่ม ผู้ใช้เปิดตู้ไม่ได้ระดับ TRL และผลการทดสอบซ้ำ
ตลาดมีลูกค้าจ่ายเงินมากพอหรือไม่ผู้ใช้จริงน้อยกว่าที่คาดการสั่งจองล่วงหน้า การทดลองขาย
การเงินรายได้พอกับต้นทุนก่อนเงินหมดหรือไม่ต้นทุนติดตั้งสูงกว่าที่คาด คืนทุนช้าการคำนวณหลายกรณี ใบเสนอราคาจริง
ทีมและการดำเนินงานทีมและพันธมิตรทำงานตามแผนได้หรือไม่สมาชิกจบการศึกษา ร้านพันธมิตรเลิกร่วมมือข้อตกลงทีม ข้อตกลงกับพันธมิตร
กฎหมายและทรัพย์สินทางปัญญาทำได้ตามกฎหมายและไม่ละเมิดสิทธิผู้อื่นหรือไม่ใช้ข้อมูลส่วนบุคคลของผู้ใช้โดยไม่ได้รับความยินยอม ชื่อบริการซ้ำกับเครื่องหมายการค้าที่จดไว้ผลการสืบค้นในบทที่ 6 ความยินยอมตามบทที่ 3

8.4.2 การให้คะแนนความเสี่ยง

เมื่อระบุความเสี่ยงได้แล้ว ทีมต้องรู้ว่าความเสี่ยงใดควรจัดการก่อน วิธีที่ใช้กันทั่วไปคือให้คะแนน 2 มิติ คือ โอกาสเกิด (Likelihood) และ ผลกระทบ (Impact) ในรายวิชานี้ใช้มาตรา 1 ถึง 3 ทั้งสองมิติ โดย 1 คือต่ำ 2 คือปานกลาง และ 3 คือสูง แล้วคูณกันเป็นคะแนนความเสี่ยง ซึ่งมีค่าตั้งแต่ 1 ถึง 9 ความเสี่ยงที่มีคะแนน 6 ขึ้นไปถือว่าสูง ต้องมีแผนรับมือก่อนเริ่มดำเนินการ ตารางที่ 8.8 แสดงตัวอย่าง

ตารางที่ 8.8 ตัวอย่างการให้คะแนนความเสี่ยงและแผนรับมือ (บริการตู้รับผ้า กรณีสมมติเพื่อการสอน)

ความเสี่ยงด้านโอกาสเกิดผลกระทบคะแนนแผนรับมือ
ผู้ใช้จริงน้อยกว่าที่คาดครึ่งหนึ่งตลาด236ทดลองที่หอพัก 1 แห่งก่อนซื้อตู้เพิ่ม
ร้านพันธมิตรเลิกร่วมมือทีมและการดำเนินงาน236ทำข้อตกลงเป็นหนังสือ และติดต่อร้านสำรอง 1 ร้าน
ระบบส่งรหัสล่มเทคโนโลยี133มีรหัสสำรองและเบอร์ติดต่อที่ตู้
ผ้าหายหรือเสียหายในตู้การเงิน122กำหนดเงื่อนไขการรับผิดชอบให้ชัด
เก็บเบอร์โทรศัพท์ผู้ใช้โดยไม่แจ้งวัตถุประสงค์กฎหมายและทรัพย์สินทางปัญญา224แจ้งวัตถุประสงค์และขอความยินยอมก่อนลงทะเบียน

ในตัวอย่าง ความเสี่ยงสูงสุด 2 ข้อไม่ใช่ด้านเทคโนโลยี แต่เป็นด้านตลาดและพันธมิตร ซึ่งสอดคล้องกับข้อสังเกตในหัวข้อ 8.2.3 ว่าส่วนที่พร้อมน้อยที่สุดของนวัตกรรมนี้คือการประสานงานกับร้าน การให้คะแนนเป็นการประมาณของทีม ไม่ใช่การวัดที่แม่นยำ ประโยชน์อยู่ที่การบังคับให้ทีมคุยกันว่าอะไรน่ากังวลที่สุด และระบุเหตุผลของคะแนนแต่ละข้อไว้ในรายงาน

8.5 การพิจารณาไปต่อ ปรับ หรือหยุด

8.5.1 จุดตัดสินในกระบวนการพัฒนา

Cooper (1990) เสนอระบบ Stage-Gate สำหรับการพัฒนาผลิตภัณฑ์ใหม่ ระบบนี้แบ่งการพัฒนาเป็นขั้นตอน และคั่นแต่ละขั้นด้วยประตู ซึ่งเป็นจุดที่ผู้บริหารพิจารณาว่าจะให้โครงการไปต่อหรือหยุด โดยใช้เกณฑ์ที่กำหนดไว้ล่วงหน้า หลักคิดสำคัญคือ การหยุดโครงการที่ไม่มีอนาคตตั้งแต่ต้นช่วยประหยัดทรัพยากรไว้ให้โครงการที่ดีกว่า บทที่ 2 ได้อธิบายกระบวนการจัดการนวัตกรรมไว้แล้ว หัวข้อนี้เน้นจุดตัดสินท้ายภาคเรียน ซึ่งเป็นส่วนที่ 8 ของแผนพัฒนาเชิงพาณิชย์

นอกจากไปต่อหรือหยุด ยังมีทางเลือกที่สาม คือ การปรับทิศทาง (Pivot) Ries (2011) ใช้คำนี้หมายถึงการเปลี่ยนแนวทางเพื่อทดสอบสมมติฐานพื้นฐานข้อใหม่ เมื่อหลักฐานแสดงว่าแนวทางเดิมไม่ได้ผล โดยยังใช้สิ่งที่เรียนรู้มาแล้ว ไม่ใช่เริ่มใหม่ทั้งหมด

8.5.2 เกณฑ์การพิจารณา

ตารางที่ 8.9 สรุปลักษณะของผลการประเมินที่มักนำไปสู่แต่ละทางเลือก เกณฑ์นี้ใช้ผลจากหัวข้อ 8.2 ถึง 8.4 ร่วมกัน ไม่ใช้ผลด้านใดด้านหนึ่งอย่างเดียว

ตารางที่ 8.9 เกณฑ์การพิจารณาไปต่อ ปรับ หรือหยุด

ทางเลือกลักษณะของผลการประเมินสิ่งที่ทีมทำต่อ
ไปต่อผู้ใช้จ่ายเงินหรือยืนยันจะจ่าย ตลาดที่เข้าถึงได้พอกับต้นทุน ความเสี่ยงสูงทุกข้อมีแผนรับมือขยายการทดลองตามแผน 6 เดือน
ปรับปัญหาของผู้ใช้มีจริง แต่ส่วนหนึ่งของโมเดลไม่ได้ผล เช่น กลุ่มลูกค้า ช่องทาง หรือรูปแบบรายได้เปลี่ยนส่วนที่ไม่ได้ผล แล้วทดสอบใหม่
หยุดไม่พบหลักฐานว่าปัญหารุนแรงพอ หรือต้นทุนสูงกว่ารายได้ในทุกกรณี และไม่มีทางปรับที่สมเหตุสมผลบันทึกสิ่งที่เรียนรู้ และพิจารณาโจทย์ใหม่

การปรับทิศทางมีได้หลายแบบ ตัวอย่างจากบริการตู้รับผ้า (กรณีสมมติเพื่อการสอน) เช่น เปลี่ยนกลุ่มลูกค้าจากผู้เช่าเป็นเจ้าของหอพัก เปลี่ยนรูปแบบรายได้จากค่าบริการต่อครั้งเป็นค่าเช่าตู้รายเดือน หรือเปลี่ยนจากตู้รับผ้าเป็นตู้รับฝากพัสดุทั่วไป ซึ่งใช้ตู้เดิมแต่ตอบปัญหาที่ใหญ่กว่า ทุกแบบใช้ความรู้ที่ทีมได้มาแล้ว และต้องทดสอบสมมติฐานใหม่ด้วยวิธีในบทที่ 4

8.5.3 การหยุดไม่ใช่ความล้มเหลวของรายวิชา

ในรายวิชานี้ กลุ่มที่สรุปว่าควรปรับหรือหยุด ไม่ได้คะแนนน้อยกว่ากลุ่มที่สรุปว่าไปต่อ รูบริกที่ 3 ประเมินว่ากลุ่มใช้กรอบการประเมินอย่างถูกต้อง มีข้อมูลที่มีที่มา และขั้นตอนถัดไปเป็นรูปธรรม ไม่ได้ประเมินว่านวัตกรรมต้องสำเร็จ ข้อสรุปว่าควรหยุดที่มีหลักฐานชัดเจน มีคุณค่ามากกว่าข้อสรุปว่าไปต่อที่ไม่มีหลักฐาน เพราะทักษะที่รายวิชาต้องการคือการตัดสินจากหลักฐาน

8.6 การรวมเป็นแผนพัฒนาเชิงพาณิชย์ฉบับสมบูรณ์

8.6.1 โครงรายงานและความเชื่อมโยงกับรูบริกที่ 3

บทที่ 7 ได้กำหนดโครงแผนพัฒนาเชิงพาณิชย์ 8 ส่วนไว้ในตารางที่ 7.13 บทนี้เพิ่มส่วนที่ 7 คือความพร้อมและความเสี่ยง จากหัวข้อ 8.2 ถึง 8.4 และส่วนที่ 8 คือข้อสรุปและก้าวต่อไป จากหัวข้อ 8.5 ตารางที่ 8.10 แสดงว่าเกณฑ์แต่ละข้อในรูบริกที่ 3 ประเมินจากส่วนใดของรายงาน และระดับสูงสุดของแต่ละเกณฑ์ต้องมีอะไร

ตารางที่ 8.10 เกณฑ์ของรูบริกที่ 3 กับส่วนของรายงาน

เกณฑ์ (น้ำหนัก)ส่วนของรายงานที่ใช้ประเมินระดับสูงกว่ามาตรฐานต้องมี
ความครบถ้วนของกระบวนการ (ร้อยละ 20)ส่วนที่ 1 ถึง 8 ทั้งฉบับครบตั้งแต่โจทย์ปัญหา แนวคิด ต้นแบบ ผลการทดสอบ ถึงแผนพัฒนาเชิงพาณิชย์ และแต่ละส่วนเชื่อมโยงกัน
การประเมินโอกาสทางนวัตกรรม (ร้อยละ 20)ส่วนที่ 3 และ 7ใช้กรอบการประเมินที่เรียน เช่น ระดับความพร้อมของเทคโนโลยี ขนาดตลาด และคู่แข่ง พร้อมข้อมูลที่มีที่มา
แผนพัฒนาเชิงพาณิชย์ (ร้อยละ 25)ส่วนที่ 5 6 และ 8ระบุโมเดลธุรกิจ ช่องทางเข้าสู่ตลาด แหล่งสนับสนุน และขั้นตอนถัดไปที่เป็นรูปธรรม
ทรัพย์สินทางปัญญาและการถ่ายโอนนวัตกรรม (ร้อยละ 15)ส่วนที่ 4ระบุประเภททรัพย์สินทางปัญญาที่เกี่ยวข้อง ผลการสืบค้นสิทธิบัตร และแนวทางการแสวงหาหรือถ่ายโอนนวัตกรรมที่เหมาะสม
การอ้างอิงและการเขียนรายงาน (ร้อยละ 10)ทั้งฉบับอ้างอิงครบทุกข้อมูลที่นำมาใช้ ภาษาวิชาการถูกต้อง
การสื่อสารในการนำเสนอ (ร้อยละ 10)การนำเสนอครั้งที่ 3ถ่ายทอดคล่องแคล่ว เป็นลำดับ ใช้สื่อดิจิทัลช่วยให้เข้าใจง่าย อยู่ในเวลา

8.6.2 การทำให้แผนเป็นเรื่องเดียวกัน

ปัญหาที่พบบ่อยที่สุดของรายงานฉบับสมบูรณ์คือแต่ละส่วนเขียนโดยสมาชิกต่างคน และใช้ข้อมูลคนละชุด เช่น ส่วนที่ 1 ระบุกลุ่มผู้ใช้เป็นนิสิตหอพัก แต่ส่วนที่ 5 ระบุกลุ่มลูกค้าเป็นคนทำงานทั่วไป หรือราคาในส่วนที่ 6 ไม่ตรงกับราคาที่ใช้คำนวณขนาดตลาดในส่วนที่ 7 ความไม่ตรงกันเหล่านี้ทำให้คะแนนเกณฑ์ความครบถ้วนของกระบวนการลดลง แม้ทุกส่วนจะเขียนครบ

วิธีป้องกันคือกำหนดตัวเลขและข้อความหลักไว้ที่เดียวก่อนเขียน ได้แก่ กลุ่มผู้ใช้แรก คุณค่าที่เสนอ ราคา จำนวนผู้ใช้ที่คาด ระดับ TRL และข้อสรุปไปต่อ ปรับ หรือหยุด แล้วให้ทุกส่วนใช้ข้อมูลชุดนี้ เมื่อมีการเปลี่ยนตัวเลขใด ให้แก้ที่ชุดข้อมูลหลักก่อน แล้วไล่แก้ทุกส่วนที่ใช้ตัวเลขนั้น หลังเขียนเสร็จ ให้สมาชิกคนหนึ่งที่ไม่ได้เขียนส่วนนั้นอ่านทั้งฉบับเพื่อตรวจความต่อเนื่อง

8.6.3 ส่วนที่ 8 ข้อสรุปและก้าวต่อไป

ส่วนที่ 8 เป็นส่วนที่กลุ่มมักเขียนสั้นและกว้างเกินไป เช่น "จะพัฒนาต่อไปให้ดียิ่งขึ้น" ซึ่งไม่ตรงกับเกณฑ์ขั้นตอนถัดไปที่เป็นรูปธรรม ส่วนนี้ควรมี 3 องค์ประกอบ องค์ประกอบแรกคือข้อสรุปไปต่อ ปรับ หรือหยุด พร้อมเหตุผลที่อ้างผลจากหัวข้อ 8.2 ถึง 8.4 องค์ประกอบที่สองคือแผน 6 เดือน ระบุกิจกรรม ผู้รับผิดชอบ และตัวชี้วัด องค์ประกอบที่สามคือทรัพยากรที่ต้องใช้ และแหล่งที่จะได้มา เช่น แหล่งสนับสนุนในบทที่ 7 ตารางที่ 8.11 แสดงตัวอย่างแผน 6 เดือน

ตารางที่ 8.11 ตัวอย่างแผน 6 เดือน (บริการตู้รับผ้า กรณีสมมติเพื่อการสอน)

เดือนกิจกรรมผู้รับผิดชอบตัวชี้วัด
1ทำข้อตกลงกับหอพัก 1 แห่งและร้านซักอบรีด 1 ร้านสมาชิกด้านพันธมิตรข้อตกลงที่ลงนามแล้ว 2 ฉบับ
2ติดตั้งตู้ 1 ตู้ และเปิดให้บริการโดยเก็บเงินจริงทั้งทีมมีการใช้อย่างน้อย 150 ครั้งในเดือนแรก
3ทำระบบส่งรหัสอัตโนมัติสมาชิกด้านเทคโนโลยีส่งรหัสได้ภายใน 1 นาทีโดยไม่มีคนช่วย
4ทบทวนตัวเลขการใช้ รายได้ และต้นทุนจริงสมาชิกด้านการเงินรายงานเทียบกับประมาณการในตารางที่ 8.5
5พิจารณาไปต่อ ปรับ หรือหยุด อีกครั้งทั้งทีมข้อสรุปพร้อมหลักฐาน
6ถ้าไปต่อ เตรียมเอกสารขอรับการสนับสนุนสมาชิกด้านเอกสารเอกสารพร้อมยื่นตามประกาศของหน่วยงาน

8.6.4 กำหนดการของงานกลุ่มครั้งที่ 3

ตามแผนการสอนใน มคอ.3 งานกลุ่มครั้งที่ 3 มีกำหนดการดังนี้ สัปดาห์ที่ 13 เรียนเนื้อหาบทนี้ สัปดาห์ที่ 14 เป็น Workshop แผนพัฒนาเชิงพาณิชย์ในชั้นเรียน ซึ่งกลุ่มควรนำร่างส่วนที่ 1 ถึง 6 มาด้วย เพื่อทำส่วนที่ 7 และ 8 พร้อมปรึกษาผู้สอน สัปดาห์ที่ 15 ปรึกษาร่างรายงานรายกลุ่ม และสัปดาห์ที่ 16 นำเสนอครั้งที่ 3 พร้อมส่งรายงานฉบับสมบูรณ์

8.7 งานวิจัยที่เกี่ยวข้อง

งานวิจัยในตารางที่ 8.12 เป็นที่มาของเครื่องมือประเมินในบทนี้ และแสดงว่าการประเมินโอกาสต้องดูทั้งเทคโนโลยีและตลาด

ตารางที่ 8.12 งานวิจัยที่เกี่ยวข้องกับการประเมินโอกาสทางนวัตกรรม

งานวิจัยประเด็นที่ศึกษาข้อค้นพบสำคัญข้อสรุปสำหรับนิสิต
Cooper (1990)ระบบ Stage-Gate สำหรับการพัฒนาผลิตภัณฑ์ใหม่ (บทความเชิงแนวคิดจากประสบการณ์ของบริษัท)แบ่งการพัฒนาเป็นขั้น คั่นด้วยจุดพิจารณาไปต่อหรือหยุดตามเกณฑ์ที่กำหนดล่วงหน้ากำหนดเกณฑ์ก่อนดูผล แล้วตัดสินตามเกณฑ์
Mankins (2009)ทบทวนความเป็นมาและการใช้ TRLTRL พัฒนาที่ NASA และหน่วยงานอื่นนำไปใช้อย่างแพร่หลายใช้ TRL เป็นภาษากลางในการบอกความพร้อม
Di Stefano และคณะ (2012)ทบทวนงานวิจัยเรื่องเทคโนโลยีผลักและตลาดดึงแนวคิดทั้งสองไม่ได้ขัดกัน นวัตกรรมเกิดจากการเชื่อมเทคโนโลยีกับความต้องการของตลาดประเมินทั้งเทคโนโลยีและตลาดเสมอ

8.8 ข้อผิดพลาดที่พบบ่อยในการประเมินโอกาสทางนวัตกรรม

ข้อผิดพลาดในตารางที่ 8.13 พบบ่อยในรายงานแผนพัฒนาเชิงพาณิชย์ และส่วนใหญ่ทำให้คะแนนเกณฑ์การประเมินโอกาสทางนวัตกรรมลดลง

ตารางที่ 8.13 ข้อผิดพลาดที่พบบ่อยในการประเมินโอกาสทางนวัตกรรม

ข้อผิดพลาดตัวอย่างวิธีแก้
ระบุระดับ TRL สูงเกินหลักฐานระบุระดับ 8 ทั้งที่ยังไม่เคยขายจริงระบุระดับที่มีหลักฐานครบตามตารางที่ 8.4
ใช้ขนาดตลาดจากบนลงล่างอย่างเดียว"ตลาดทั้งประเทศมีมูลค่าหลายพันล้านบาท"ประมาณจากล่างขึ้นบนพร้อมที่มาของตัวเลข
เขียนว่าไม่มีคู่แข่ง"ยังไม่มีใครทำบริการนี้"เปรียบเทียบกับทางเลือกเดิมตามตารางที่ 8.6
ระบุความเสี่ยงด้านเดียวมีแต่ความเสี่ยงด้านเทคโนโลยีตรวจครบ 5 ด้านตามตารางที่ 8.7
สรุปไปต่อโดยไม่มีเกณฑ์"ทีมเชื่อว่าจะประสบความสำเร็จ"ใช้เกณฑ์ในตารางที่ 8.9 และอ้างผลการประเมิน
ตัวเลขแต่ละส่วนไม่ตรงกันราคาในส่วนที่ 6 ต่างจากส่วนที่ 7ใช้ชุดข้อมูลหลักชุดเดียวตามหัวข้อ 8.6.2
ก้าวต่อไปกว้างเกินไป"จะพัฒนาให้ดียิ่งขึ้น"เขียนแผน 6 เดือนพร้อมตัวชี้วัดตามตารางที่ 8.11
กรณีศึกษา: เครื่องอบแห้งพลังงานแสงอาทิตย์ของกลุ่มเกษตรกร (กรณีสมมติเพื่อการสอน)

คณะวิศวกรรมศาสตร์ของมหาวิทยาลัยแห่งหนึ่งพัฒนาเครื่องอบแห้งผลไม้พลังงานแสงอาทิตย์ ผ่านการทดสอบในแปลงสาธิตของมหาวิทยาลัย 1 ฤดูกาล กลุ่มนิสิตในรายวิชาได้รับโจทย์ให้ประเมินว่าควรนำเครื่องนี้ไปให้กลุ่มเกษตรกรผู้ปลูกกล้วยในจังหวัดใช้หรือไม่

กลุ่มระบุว่าเทคโนโลยีอยู่ระดับ 6 ตามถ้อยคำปรับในตารางที่ 8.3 เพราะทำงานได้ครบแต่ยังไม่เคยใช้ในพื้นที่ของเกษตรกร การประมาณตลาดแบบจากล่างขึ้นบนจากรายชื่อสมาชิกกลุ่มเกษตรกรพบว่าสมาชิกที่แปรรูปกล้วยตากเองมีไม่มาก และทางเลือกเดิมคือการตากแดดบนตะแกรงซึ่งแทบไม่มีต้นทุน ความเสี่ยงสูงสุดคือด้านการเงิน เพราะราคาเครื่องสูงเมื่อเทียบกับรายได้จากกล้วยตากของสมาชิกแต่ละราย

กลุ่มสรุปให้ปรับทิศทาง โดยเปลี่ยนกลุ่มลูกค้าจากเกษตรกรรายบุคคลเป็นวิสาหกิจชุมชนที่ใช้เครื่องร่วมกัน และเสนอแผนทดลองใช้ 1 เครื่องในฤดูกาลถัดไป พร้อมบันทึกคุณภาพสินค้าและเวลาที่ประหยัดได้เทียบกับการตากแดด

คำถามชวนคิด ข้อสรุปของกลุ่มอ้างผลการประเมินด้านใดบ้าง และถ้าผลการทดลองในฤดูกาลถัดไปไม่ดีขึ้น กลุ่มควรพิจารณาทางเลือกใด

บทสรุป

เมื่อศึกษาบทนี้แล้ว นิสิตสามารถอธิบายได้ว่านวัตกรรมเริ่มได้ทั้งจากเทคโนโลยีผลักและตลาดดึง และนวัตกรรมที่สำเร็จต้องเชื่อมเทคโนโลยีเข้ากับความต้องการของตลาด นิสิตระบุแหล่งของโอกาสจากเทคโนโลยี เช่น เทคโนโลยีที่ราคาลดลง สิทธิบัตรที่หมดอายุ และผลงานวิจัยที่ยังไม่มีผู้นำไปใช้ และระบุระดับความพร้อมของเทคโนโลยีของนวัตกรรมของกลุ่มจากหลักฐาน โดยรู้ว่า TRL วัดเฉพาะความพร้อมของเทคโนโลยี

นิสิตคำนวณขนาดตลาดแบบจากล่างขึ้นบนพร้อมที่มาของตัวเลข เปรียบเทียบนวัตกรรมกับคู่แข่งและทางเลือกเดิม จำแนกความเสี่ยง 5 ด้าน และคำนวณคะแนนความเสี่ยงจากโอกาสเกิดและผลกระทบได้ นิสิตใช้เกณฑ์การพิจารณาไปต่อ ปรับ หรือหยุด กับนวัตกรรมของกลุ่ม และใช้โครงแผนพัฒนาเชิงพาณิชย์ร่วมกับรูบริกที่ 3 ตรวจความครบถ้วนและความเชื่อมโยงของรายงานฉบับสมบูรณ์ ซึ่งเป็นผลงานรวมของรายวิชาทั้งภาคเรียน

ศัพท์และแนวคิดสำคัญประจำบท
  • เทคโนโลยีผลัก (Technology push)
  • ตลาดดึง (Market pull)
  • ระดับความพร้อมของเทคโนโลยี (Technology Readiness Level)
  • การประมาณจากบนลงล่าง (Top-down)
  • การประมาณจากล่างขึ้นบน (Bottom-up)
  • ทางเลือกเดิม (Existing alternatives)
  • โอกาสเกิด (Likelihood)
  • ผลกระทบ (Impact)
  • ระบบ Stage-Gate (Stage-Gate system)
  • การปรับทิศทาง (Pivot)
  • แผนพัฒนาเชิงพาณิชย์ (Commercialization plan)

คำถามทบทวน

  1. จงเปรียบเทียบนวัตกรรมแบบเทคโนโลยีผลักกับตลาดดึง และบอกว่านวัตกรรมของกลุ่มเริ่มจากทางใด
  2. จงยกตัวอย่างแหล่งของโอกาสทางนวัตกรรมจากเทคโนโลยี 3 แหล่ง พร้อมตัวอย่างที่ใช้กับธุรกิจขนาดเล็กได้
  3. ถ้านวัตกรรมมีต้นแบบที่ทำงานได้ครบทุกหน้าที่ และทดสอบกับผู้ใช้จริงแล้ว แต่ยังไม่เคยขาย ควรอยู่ระดับ TRL ใดตามถ้อยคำปรับในตารางที่ 8.3 เพราะเหตุใด
  4. เพราะเหตุใด TRL อย่างเดียวจึงไม่พอสำหรับการประเมินโอกาสทางนวัตกรรม
  5. ร้านกาแฟแห่งหนึ่งอยู่ใกล้สำนักงาน 25 แห่ง แต่ละแห่งมีพนักงาน 40 คน ร้อยละ 20 ของพนักงานสนใจบริการสั่งกาแฟล่วงหน้า และแต่ละคนสั่งเดือนละ 10 แก้ว แก้วละ 50 บาท จงคำนวณตลาดที่เข้าถึงได้ต่อเดือน
  6. ความเสี่ยงที่มีโอกาสเกิด 3 และผลกระทบ 2 มีคะแนนเท่าใด และต้องมีแผนรับมือก่อนเริ่มดำเนินการหรือไม่ตามเกณฑ์ในบทนี้
  7. การปรับทิศทางต่างจากการหยุดอย่างไร จงยกตัวอย่างการปรับทิศทาง 2 แบบ
  8. จงอธิบายวิธีทำให้รายงานแผนพัฒนาเชิงพาณิชย์ที่เขียนโดยสมาชิกหลายคนมีข้อมูลตรงกันทั้งฉบับ
งานของกลุ่ม

กิจกรรมท้ายบท: งานกลุ่มครั้งที่ 3 แผนพัฒนาเชิงพาณิชย์ของนวัตกรรมและรายงานฉบับสมบูรณ์

ให้ทำเป็นกลุ่ม นำร่างมาทำต่อใน Workshop สัปดาห์ที่ 14 นำเสนอและส่งรายงานฉบับสมบูรณ์ในสัปดาห์ที่ 16 งานนี้ประเมินด้วยรูบริกที่ 3 คิดเป็นร้อยละ 25 ของคะแนนรายวิชา

  1. ระบุระดับความพร้อมของเทคโนโลยีของแต่ละส่วนของนวัตกรรมตามแบบในตารางที่ 8.4 พร้อมหลักฐาน
  2. ประมาณขนาดตลาดแบบจากล่างขึ้นบนตามแบบในตารางที่ 8.5 ระบุที่มาของตัวเลขทุกตัว และคำนวณกรณีที่แย่กว่าที่คาดด้วย
  3. เปรียบเทียบนวัตกรรมกับคู่แข่งและทางเลือกเดิมอย่างน้อย 3 ทางเลือกตามแบบในตารางที่ 8.6
  4. ระบุความเสี่ยงครบ 5 ด้าน ให้คะแนน และเขียนแผนรับมือความเสี่ยงที่มีคะแนน 6 ขึ้นไปตามแบบในตารางที่ 8.8
  5. สรุปว่าควรไปต่อ ปรับ หรือหยุด ตามเกณฑ์ในตารางที่ 8.9 และเขียนแผน 6 เดือนตามแบบในตารางที่ 8.11
  6. รวมผลงานจากกิจกรรมท้ายบทที่ 5 ถึง 7 และงานกลุ่มครั้งที่ 1 และ 2 เป็นรายงานฉบับสมบูรณ์ตามโครงในตารางที่ 7.13
  7. ตรวจรายงานด้วยแบบตรวจในตารางที่ 8.14 ก่อนส่ง

ตารางที่ 8.14 แบบตรวจความครบถ้วนของรายงานฉบับสมบูรณ์ก่อนส่ง

รายการตรวจเกณฑ์ในรูบริกที่ 3ผ่านส่วนที่ต้องแก้
มีครบ 8 ส่วนตามตารางที่ 7.13ความครบถ้วนของกระบวนการ
กลุ่มผู้ใช้ ราคา และตัวเลขหลักตรงกันทุกส่วนความครบถ้วนของกระบวนการ
มี TRL ขนาดตลาด และคู่แข่ง พร้อมที่มาของข้อมูลการประเมินโอกาสทางนวัตกรรม
มีโมเดลธุรกิจ ช่องทาง แหล่งสนับสนุน และแผน 6 เดือนแผนพัฒนาเชิงพาณิชย์
มีประเภททรัพย์สินทางปัญญาและผลการสืบค้นสิทธิบัตรทรัพย์สินทางปัญญาและการถ่ายโอนนวัตกรรม
อ้างอิงครบทุกข้อมูลที่นำมาใช้การอ้างอิงและการเขียนรายงาน
